  EXP1 ! DUP 0< IF SIGN1 ON THEN M! ( save exp & sign)

( GRAPHSORT - XRC                                      9-19-85)
: XC ( n n' -- )
\ Exchange column n with row-column n'.
  #ROWZ 0 DO DUP    I A@
             3 PICK I A@  ( n n' n'v nv --)
             3 PICK I A!  ( n n' nv --)
             3 PICK I A!
          LOOP 2DROP ;
: XR ( n n' -- )
\ Exchange row n with row-column n'.
  #COLZ 0 DO I 2 PICK A@
             I 4 PICK A@  ( n n' n'v nv --)
             I 4 PICK A!  ( n n' nv --)
             I 4 PICK A!
          LOOP 2DROP ;
: XRC ( n n' --) 2DUP XR XC ;
