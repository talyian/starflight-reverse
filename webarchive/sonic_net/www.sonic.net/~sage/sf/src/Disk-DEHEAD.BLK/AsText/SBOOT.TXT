: .#LINES ( -- \ plot current string as center formated)
  #YBLT
  $ADDR @ SUB$ADDR !
  BEGIN
    SUB$ADDR @ CPL @ + DUP
    $ADDR @ $LEN @ + U<
  WHILE
    ?BL<-                     ( address.of.linebreak.blank --)
    SUB$ADDR @ -
    SUB$LEN ! .SUB$
    CHARHEIGHT @ LINESPACING @ + NEGATE YBLT +!
    SUB$LEN @ 1+ SUB$ADDR +!
  REPEAT DROP
  $ADDR @ $LEN @ + SUB$ADDR @ - SUB$LEN ! .SUB$ ;
