( PCOMP -  COPY-PCR                                      3-8-85)

: COPY-PCR ( --- \ copy current phrase control variables)
   ( to phrase control record)
    (LP) @ LP C! (LSC) @ LSC C!
    (FC) @ FC C! (DEDLN) @ DEDLN C!
    (DEDLY) @ DEDLY C! (UA) @ UA C! ;









---- Screen 14 ----
( PCOMP - P" SAVE-PHRASE                                4/22/85)

: P" ( <phrase> --- \ get phrase and store it)
    ASCII " WORD COUNT  2DUP >UPPERCASE  ( addr count --)
    >HUFF 'SAVE-PHRASE @ EXECUTE ;
IMMEDIATE

: SAVE-PHRASE ( addr count -- \ save phrase under current sub)
   (SUBJ) 1.5@ >C SET-CURRENT
   IOPEN PCR-CLASS 0 1 *CREATE
   COPY-PCR
   DUP 1+ VICREATE CI IINSERT
   IOPEN DUP PHR-CNT C! PHRASE SWAP CMOVE
   CDROP CDROP ICLOSE ;


---- Screen 15 ----
( PCOMP - SAVE-SYLB ORIG[ ]ORIG                          3-8-85)

: SAVE-SYBL ( addr count -- \ save syllable in vst)
  (VST) 1.5@ >C SET-CURRENT
  DUP 1+ VICREATE CI IINSERT
  IOPEN DUP PHR-CNT C! PHRASE SWAP CMOVE
  CDROP ICLOSE ;

: ORIG[ ( ( --i) , -- \ put current originator onto context stk)
  (ORIG) 1.5@ >C SET-CURRENT ;

: ]ORIG ( ( --i) , -- \ drop current originator )
  ICLOSE ;

( PCOMP - BEGIN-OPR-ASSEMBLY END-OPR-ASSEMBLY            3-8-85)

: BEGIN-OPR-ASSEMBLY ( --- \ preface phrase text with this)
    IROOT >C SET-CURRENT
    IOPEN  7 0 IFIND DROP ( look for iapplication)
    11 36 ICREATE 2DUP CI IINSERT (ALLPHR) 1.5!
    ' SAVE-PHRASE 'SAVE-PHRASE ! ;

: END-OPR-ASSEMBLY ( --- \ follows phrase text)
  SAVE-BUFFERS ;

\ BEGIN-OPR-ASSEMBLY MUST
\ BE PLACED ON THE FIRST SCREEN OF
\ THE ORIGINATOR RECORD DESCRIPTIONS

( PCOMP - BEGIN-VST END-VST                             3-11-85)

: BEGIN-VST ( --- \ start vocabulary syntax table)
    11 48 ICREATE 2DUP (VST) 1.5!
    (ORIG) 1.5@ IINSERT
    ORIG[ (VST) 1.5@ VST-PTR 1.5! ]ORIG
    ' SAVE-SYBL 'SAVE-PHRASE ! ;

: END-VST ( --- \ return to general phrases)
    ' SAVE-PHRASE 'SAVE-PHRASE ! ;

( PCOMP - ORGINATOR: SUBJECT:                            3-8-85)

: ORIGINATOR: ( <name> -- \ create a new originator record)
  OPR-CLASS
  [COMPILE] ' EXECUTE   ( n -- \ get orginator number)
  ICREATE 2DUP          ( i i --)
  (ORIG) 1.5!           ( -- \ make current orginator)
  (ALLPHR) 1.5@ IINSERT ;

: SUBJECT: ( <name> -- \ create a new subject record)
  SUBJ-CLASS
  [COMPILE] ' EXECUTE   ( n -- \ get subject number)
  ICREATE 2DUP          ( i i --)
  (SUBJ) 1.5!           ( -- \ make current subject)
  (ORIG) 1.5@ IINSERT ;

( PCOMP - SPACE HOMEWORLD LEGAL-SPACIAL-CONTEXTS:       3-11-85)

: SPACE ( -- \ set space context bit)
  (LSC) @ SP OR (LSC) ! ;

: HOMEWORLD ( -- \ set homeworld context bit)
  (LSC) @ HW OR (LSC) ! ;

: LEGAL-SPACIAL-CONTEXTS: ( -- \ reset legal spacial context)
  (LSC) OFF ;

---- Screen 25 ----
( PCOMP - LEGAL-POSTURES: OBSEQUIOUS FRIENDLY ...       3-11-85)

: LEGAL-POSTURES: ( -- \ reset legal posture byte)
  (LP) OFF ;

: OBSEQUIOUS ( -- \ set obseq bit in lp byte)
  OBSEQ OR(LP) ;

: FRIENDLY ( -- \ set friendly bit in lp byte)
  FRIEND OR(LP) ;

: NEUTRAL ( -- \ set diplomatic bit in lp byte)
  NEUT. OR(LP) ;

: HOSTILE ( -- \ set hostile bit in lp byte)
  HOST. OR(LP) ;
