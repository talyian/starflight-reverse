HEAD: BUYPOD ( -- )
T: %PODS @ +BIT 16 < 500. m?BALANCE AND
 IF mD@BALANCE 500. D- mD!BALANCE BALANCEMESS
   1 16 0 DO DUP %PODS @ AND 0=
     IF %PODS +! 500 %CARGO +! LEAVE ELSE 2* THEN
   LOOP m!TFLAG .PODS ?PODS
   ELSE 1 OMESS %PODS @ +BIT 16 =
   IF 1 28 CMESS ." NO CARGO POD BAYS AVAILABLE." !CRS
     ELSE 0$$$MESS
 THEN KEY'TIL QUITMESS THEN T;

( Ship Configuration Buy Ship Parts               AWK 09/21/85 )
HEAD: GET-CLASS ( -- )
T: BEGIN XYSCAN 0=
   IF -1 * NCRS +! NCRS @ OCRS @ = NOT NCRS @ 0 5 WITHIN AND
     IF BLT 96 NCRS @ DUP OCRS ! 10 * - yBLT ! BLT
       ELSE OCRS @ NCRS !
     THEN 0
   ELSE DROP 1 THEN
 ?TRIG IF DROP BUYPART THEN UNTIL XORMODE ON FQUIT OFF T;

HEAD: GET-PART ( -- )
T: OCRS @ ?DUP IF DUP #PART ! ?GOTPART
   IF 1 OMESS 1 15 CMESS ." DE-EQUIP FIRST." !CRS
     KEY'TIL QUITMESS   ELSE @CRS 14 96 POS. 29 wBLT !
     NCRS OFF OCRS OFF XORMODE ON CRSCOLOR @ !COLOR
 BLT GET-CLASS BLT !CRS THEN ELSE BUYPOD THEN T;
---- Screen 28 ----
( Ship Configuration Sell Ship Parts              AWK 09/21/85 )

HEAD: SELLPOD ( -- )
T: %CARGO @ 499 > %PODS @ AND
 IF 1 16 DO
     I BIT %PODS @ AND %CARGO @ 499 > *
     IF I BIT -1 * %PODS +! 500. mD@BALANCE D+ mD!BALANCE
       .PODS ?PODS BALANCEMESS m!TFLAG LEAVE -500 %CARGO +!
     THEN
     -1 +LOOP
   ELSE %PODS @
     IF 1 OMESS 1 17 CMESS ." POD IS NOT EMPTY." !CRS
       KEY'TIL QUITMESS
     THEN
 THEN T;

---- Screen 29 ----
( Ship Configuration Sell Ship Parts              AWK 10/23/85 )

HEAD: PUT-PART ( -- )
T: OCRS @ ?DUP
 IF SALEVALUE D@ OR
   IF PARTADDR OFF DROP 0 OCRS @ !POINTS
     SALEVALUE D@ mD@BALANCE D+ mD!BALANCE m!TFLAG
     .SHIP .CLASS BALANCEMESS OCRS @ .VALUE
   ELSE DROP THEN
 ELSE SELLPOD OCRS @ .VALUE THEN T;

HEAD: (REPAIR-SHIP) ( d -- )
T: mD@BALANCE 2SWAP D- mD!BALANCE m!TFLAG
 14 0 DO 100 %ARMOR I + C! 2 +LOOP
 %SHIELD 1+ C@ 7 AND 500 *      %SHIELD-CNT !
 %ARMOR  1+ C@ 7 AND 500 * 250 + %ARMOR-CNT  ! T;

HEAD: REPAIR-SHIP ( -- )
T: 0 OMESS 1 OMESS REPAIR-COST 2DUP OR

   IF 2DUP ?REPAIR

   ELSE .REPAIR-COST 1 18 CMESS .0$$$ !CRS KEY'TIL THEN
   ELSE 2DROP 0 22 CMESS ." NO REPAIRS ARE NEEDED." !CRS
 KEY'TIL THEN 1 OMESS BALANCEMESS T;
