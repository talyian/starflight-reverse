\ start of roster stuff                           rfg14may85)
head: raddr \ index --- addr | within roster
t:     3* roster + t;

head: rclear \ clear roster array
t:  roster 18 0 fill t;

head: rinstall \ iaddr --- |install into array if not there
t:  6 0 do i raddr           \ cell address
         1.5@ 2over d=     \ already there?
          if  leave
          else i raddr 1.5@ or 0=         \ empty slot
           if 2dup i raddr 1.5! leave then
          then
      loop 2drop t;        \  no extra stack value, even if err

---- Screen 4 ----
\ more roster: rdelete, role-call                 rfg29jun85)

: role-call
  rclear
 *assign-crew >c set-current
         inst-sib 17 +      \ first address
  6 0 do i 3* over + 1.5@ >c set-current
        ^vit c@ if ci  rinstall then iclose
         loop drop iclose ;

: ?all-smthng \ 0 or 100 --- | check if all same condition
 >r 1 roster

    if  >c set-current  ^vit c@ iclose j =

\ heal each crewmember                            rfg29jun85)
head: (heal)
t:    role-call \ refresh- other processes may change crewlist
  ?all-well not
   if 6 0 do i raddr 1.5@ 2dup or        \ anybody in this slot
           if >c set-current
              ^vit c@ ?dup             \ don't do if dead
               if ^sta c@ 3 >          \ under treatment
                  if 2+ else 1+ then
                 100 min ^vit c!
               then iclose            \ crewmember
            else 2drop
            then
           loop
    then medskill healtime ! t; \ new healing rate

: .vits
xormode @ >r xormode off color @ >r
  white !color vitwindow 0 1 wpos
   6 0 do gcr wchars @ black poly-erase-text
           i raddr  1.5@ 2dup
           >c set-current or
             if ^name $.
              14 ^name c@ - ?dup
               if 0 do ." ." loop then
              4 black poly-erase-text
              ^vit c@ 3 .r ." %"
             then iclose
        loop r> !color r> xormode ! ;

\ highpt  + sweepstk                              rfg06jun85)

v: highpt \ highest value stored
v: winner 1 allot \ iaddr of one with highest rating

head: sweepstk \ field index --- | set highpt and winner
t:    0 highpt !  \ before first pass
  *assign-crew >c set-current captain 1.5@ winner 1.5!
              inst-sib 17 +
  6 0 do i 3* over + 1.5@ >c set-current \ which crewmember
         over >skill c@ dup highpt @ > \ get field, then value
         if ci winner 1.5! highpt ! \ new winner
         else drop \ skill
         then iclose                     \ crewmember
      loop 2drop iclose t; \ *assign

head: obitcapt \ special case for captain->science officer
t:   *assign-crew >c set-current
    captain 1.5@ >c set-current
     ^vit c@ 0=

        captain 1.5! iclose
     then cdrop iclose t;

head: clrmemb t: ^sci 6 0 fill t;

\ general case replacement                        rfg14may85)
head: (obits)
t:  *assign-crew >c+s captain
  6 1 do  i 3* over + 1.5@ >c+s \ each crewmember
          ^vit c@ 0=                 \ is deceased
          if clrmemb iclose  i sweepstk winner 1.5@
            i 3* 4 pick + 1.5!
          else iclose then                \ crewmember
      loop drop iclose t;  \ *assign
