( 'Flight Button tasking                         AWK 12/21/84 )
EXIT
: XEQ-BUTTON ( n -- )    'BUTTON @ EXECUTE ;

V= BTN-REC#

: .ABTN ( menu# btn# -- )
 >R 58 FILE# ! RECORD# ! 93 119 I 11 * - POS. >2FONT
 vBLT @ 93 OVER 7 - 156 BLACK POLY-WINDOW-FILL
 WHITE !COLOR 1BTN R> 12 * + 12 TYPE SET-CURRENT ;

: .BTN-TEXT ( menu# -- )
 58 FILE# ! DUP RECORD# ! BTN-REC# !
 BLACK ERASE-BUTTONS 93 119 POS. >2FONT WHITE !COLOR
 #BTN C@ 12 * 0 DO 1BTN I + 12 TYPE NEXT-BUTTON 12 +LOOP
 SET-CURRENT ;

( 'Flight Button tasking                         AWK 12/21/84 )
EXIT
: NEW-BUTTON ( n -- )
 58 FILE# ! BTN-REC# @ RECORD# !
 THIS-BUTTON @ SWAP - DUP 0 #BTN C@ WITHIN
 IF BLUE OVER BLACK THIS-BUTTON @ .HIGHLIGHT .HIGHLIGHT
 THIS-BUTTON ! ELSE DROP THEN SET-CURRENT ;









---- Screen 5 ----
( 'Flight Button tasking                         AWK 12/21/84 )
: BTN-CLEANER ( -- )
 V> 'BUTTON !          \ restore button function table addr
 V> DUP THIS-BUTTON !  \ restore button that called this task
 V> DUP BTN-REC# !     \ restore button menu species
 CLR-BUTTONS           \ clear old button highlights
 CTERASE DISEMBARK @ 0=
 IF .BTN-TEXT BLUE SWAP .HIGHLIGHT   \ restore button graphics
 ELSE 2DROP THEN ;     \ or not

: BTN-INITER ( addr n -- )
 BTN-REC# @ >V         \ push button menu species
 THIS-BUTTON @ >V      \ push calling button
 'BUTTON @ >V          \ push button function table
 .BTN-TEXT INIT-BUTTON ~~~~execute new button graphics
 'BUTTON ! ;               store button function table addr

: (BTN) ( -- ) THIS-BUTTON @ .ON ;
: >BTN BLUE  (BTN) ;
: BTN> BLACK (BTN) ;

: BTN-KEY-CASE ( -- )
 (XYSCAN) DROP ?DUP
 IF NEW-BUTTON THEN ?TRIG
 IF >BTN THIS-BUTTON @ XEQ-BUTTON THEN ;

\ if trigger has been pressed, the current button is executed
\ if DISEMBARK has been set, exit the starship
