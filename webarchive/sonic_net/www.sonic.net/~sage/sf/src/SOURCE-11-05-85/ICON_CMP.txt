EXIT
V= BLTOFFSET V= BLTBLK#
: NEXT-LINE  64 BLTOFFSET +! BLTOFFSET @ 1024 =
  IF 64 BLTOFFSET ! 1 BLTBLK# +! THEN ;
: BLKAD BLTBLK# @ BLOCK BLTOFFSET @ + ;
: .BLTPARMS
  NEXT-LINE " COLOR:" BLKAD SWAP CMOVE
  IOPEN BLT-COLOR C@ 0 <# # # # #> BLKAD 8 + SWAP CMOVE
  " IMAGE:" BLKAD 16 + SWAP CMOVE
  8 0 do i blt-image + c@ 0 <# # # # #> BLKAD 25 + I
      5 * + SWAP CMOVE loop ;

   BLT-NAME BLKAD 12 + 16 CMOVE
    ?child if .bltparms ?child if .bltparms iclose then
              iclose then ;
: convertblts ( blk# -- \ convert blts to compiler format)
  bltblk# !  64 bltoffset !
  open-blt-box
  ' cblt all ;

: make-printable ( blk# -- \ convert non-printable to blanks)
  1024 0 do dup block i + dup c@ bl <
            if bl swap c! else drop then loop  drop ;
: make-print ( from to -- )
  1+ swap do i make-printable loop ;

: ICON-NAME: ( <name> -- \ get icon-name from input stream)
  ( and save if scale#=1)
  1 ICON# +!
  2ND-COLOR OFF
  ASCII " WORD
  SCALE# @ 1 = ICON# @ 27 < AND ( 1st scale & 1st 27 only)
  IF DUP C@ >R 1+
      65 ICON# @ @RECORD R> CMOVE
  ELSE DROP THEN ;

: COLOR: ( <color-code> -- \ get color code & save for current)
  ( icon)
  GET-INTEGER   SCALE-FILE# @ ICON# @ @RECORD
  2ND-COLOR @ IF 9 + THEN C! ;
