T: >1FONT WHITE !COLOR 42 110 POS. ." THERE ARE NO LIVING"
 40 100 POS. ." CREWMEMBERS ON FILE." T;

HEAD: SEARCH-FOR-CREW ( -- f )
T: IFIRST 0 25 0 DO ^STA C@ 3 AND 3 XOR 2 =
 IF 1+ LEAVE ELSE INEXT THEN LOOP T;
