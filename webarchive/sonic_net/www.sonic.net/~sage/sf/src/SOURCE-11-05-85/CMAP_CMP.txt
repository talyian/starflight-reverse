---- Screen 9 ----
( CMAP - Contour map color table compiler source         3-6-85)
V= BASE-RECORD

: RGB ( -- \ set base record pointer to rgb area)
  BASE-RECORD OFF ;
: COMPOSITE ( -- \ set base record pointer to composite area)
  5 BASE-RECORD ! ;

: ADDTILE ( c c n -- \ add contour tile to cmap buffer)
  8 * CMAP + TILE-PTR !
  2 0 DO I 4 * TILE-PTR @ + >R
         >R >R
         I J C! I J 1+ C!  I' J 2+ C!  I' J 3 + C!
         R> R> R> DROP LOOP 2DROP ;


---- Screen 10 ----
( CMAP - Contour map color table compiler source         3-6-85)

: ADDRECORD ( n -- \ copy cmap to record n of cmap file)
  FILE: CMAP LITERAL SWAP @RECORD CMAP SWAP 64 CMOVE ;

: !COLORMAP ( c c c c c -- \ save color constants in cmap file)
  0 4 DO  ( for each planet type)
  0 7 DO  ( for each contour)
      I ADDTILE
      -1 +LOOP
      I BASE-RECORD @ + ADDRECORD
      -1 +LOOP ;
