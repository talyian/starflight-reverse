26 C= DIRBASE ( base block of directory)

: DOFFBLK ( dirrec# -- offset blk \ calc directory record#)
  ( offset & block)
  DUP 96 < NOT
  IF 96  DIRBASE 2+
  ELSE DUP 48 < NOT
       IF 48 DIRBASE 1+
       ELSE  0 DIRBASE
       THEN
  THEN
  >R - 21 * R> ;

: INIT-DIR ( -- \ initialize disk file directory)
  140 0 DO I FILE-NAME 12 BL FILL
           I FILE-TYPE  9  0 FILL LOOP ;
: .RJ ( n -- \ right justify) 0 5 D.R SPACE ;
: DIR ( start -- \ display directory from start to end)
  CR ." dir#             disk  #rec   rlen   slen"
  ."   start    end " 140 SWAP
  DO  CR ?TERMINAL IF LEAVE THEN
      I .RJ
      I FILE-NAME 12 TYPE
      I FILE-TYPE C@ .RJ
      I FILE-#REC  @ .RJ
      I FILE-RLEN C@ .RJ
      I FILE-SLEN C@ .RJ
      I FILE-START @ .RJ I FILE-END @ .RJ LOOP ;

: GET# ( <number> -- n \ get number from input stream)
  BL WORD NUMBER DROP ;

CASE DISK-CODE ( c -- n)
  ASCII A IS 1
  ASCII B IS 2
  ASCII X IS 3
OTHERS 0

: FILE ( file# <name> disk #rec rlen slen -- \)
       ( enter a file in the file directory)
  GET# >R BL WORD COUNT  I FILE-NAME SWAP CMOVE
  BL WORD COUNT DROP C@  DISK-CODE I FILE-TYPE C!
  GET# I FILE-#REC !     GET# I FILE-RLEN  C!
  GET# I FILE-SLEN C! R> DROP ;

: .FREE-SPACE ( -- \ display amount of free space on the disks)

  FINAL-DPB DPB @ - CR ." DISK B = " U. ."  FREE SEGS" ;

  ( of block)
  0 64 U/MOD DROP NEGATE 64 + 16* ;

: SEGISTBLK ( dp-vsa -- dp-vsa #segslstblk #rlstb #rec rlen)
  ( \ given a freespace address and the current file in )
  ( file#, calc some parameters)
  DUP BYTES>BLOCKEND          ( dp-vsa c>eob --)
  FILE# @ FILE-RLEN C@      ( dp-vsa c>eob rlen -- )
  2DUP < IF >R DROP DUP BYTES>BLOCKEND 16/ + 1024 R> THEN
  >R I /                      ( dp-vsa recs>eob -- rlen)
  FILE# @ FILE-#REC @ >R I MIN
                              ( dp-vsa #recslstblk -- #rec rlen)
  >R I J * 16 /MOD SWAP IF 1+ THEN
         ( dp-vsa #segslstblk -- #rlstblk #rec rlen)
  I I' = NOT IF OVER BYTES>BLOCKEND 16/ MAX THEN
  R> R> R> ;

( FILES - FILE-START/END                               3-4-85)

: FILE-START/END ( dp-vsa -- start-vsa end-vsa \ given a)
  ( freespace address and the current file in file#, compute)
  ( the adjusted freespace pointer and the start & end vsas.)
  SEGISTBLK ( dp-vsa #segslstblk #rlstb #rec rlen --)
  >R SWAP - ( dp-vsa #segslstblk #rec' -- rlen )
  1024 I / ( dp-vsa #segslstblk #rec' rec/blk -- rlen )
  /MOD 64 * ( dp-vsa #segslstblk remrec sblk -- rlen )
  SWAP R> * ( dp-vsa #segslstblk sblk #rbytes -- )
  16 /MOD SWAP IF 1+ THEN + + ( dp-vsa vsa-len --)
  OVER + 1- ;
