( INSTALL - LFCLAIM AINSTALL BINSTALL                   2-4-85)
EXIT
: LFCLAIM
  ( #segs -- seg \ claim space at end of avail mem)
  NEGATE LFSEG @ + DUP LFSEG ! ;

: AINSTALL
  ( array-pfa -- install indexed array in memory)
  DUP 'ARRAY ! #BYTES #ROWZ 2* + 16/ 1+ LFCLAIM OVER 6 + !
  !OFFSETS ;

: BINSTALL
  ( base-ptr-pfa #segs -- \ install un-indexed array)
  LFCLAIM SWAP ! ;

( INSTALL - CONFIGURE-SYSTEM                            2-5-85)
EXIT HEX
: CONFIGURE-SYSTEM ( customize to environment)
  12 0 SYSCALL          ( -- #K in system)
  40 *                  ( -- #seg in system)
  1- LFSEG !            ( -- \ mark last free seg)
  AINSTALLS             ( -- \ install indexed arrays)
  BINSTALLS             ( -- \ install un-indexed arrays)
  LFSEG @ @DS 1000 + -  ( -- #cache-segs)
  41 /                  ( -- #cache-blks)
  #CACHE !  AUTO-CACHE  ( -- \ set up auto cache)
  B800 DBUF-SEG !       ( -- \ screen display buffer)
  BUFFER-HEAD @ SEG>ADDR DUP 1BUFADR ! ( disk buffers)
  @ SEG>ADDR 2BUFADR ! ICINIT ( INSTALLXKEY) ;

( MEMORY - INSTANCE OP: IRESET support               1-15-85)

: MAXSPACE 3. VA>BUF ;

: INIT-SPACE ( -- ) IROOT !NEWSPACE
  150000. MAXSPACE 1.5! ;
: NEXT-NEW ( class -- d  \ allocate new space and ignore)
  ( fragments and block boundaries - for ireset)
  >R I FILE-SLEN C@ >R   ( -- slen class )
  @NEWSPACE 2DUP I IHEADLEN + M+ !NEWSPACE ( d -- slen class)
  >C CI VA>BUF          ( adr -- slen class)
  9 + I' SWAP C!        ( -- slen class \ store class)
  R> R> 2DROP
  SET-CURRENT NULLPOINTERS CI ICLOSE ;

: IRESET ( -- \ initialize object handling system)
  ICINIT CCLR VCLR
  ( MOUNT-B ) INIT-SPACE          ( initialize newspace)
  2 NEXT-NEW >C SET-CURRENT CI INST-SIB 1.5! CI INST-PREV 1.5!
  7 NEXT-NEW CI IINSERT   ( IAPPLICATION)
  3 NEXT-NEW CI IINSERT IOPEN ( ISYSTEM)
  5 NEXT-NEW 2DUP ' FRAGMENT 2! CI IINSERT ( FRAGMENT)
  4 NEXT-NEW 2DUP ' INACTIVE 2! CI IINSERT ( INACTIVE)
  SAVE-BUFFERS CCLR VCLR ;
