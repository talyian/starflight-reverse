\ crew-damage                                      (rfg18oct85)

head: crew-damage \ damage --- \ apply to crew memebers
t: #crew-damage ! init-wound
   roster dup 18 + swap
  do i 1.5@ 2dup or
     if ?injure
         if >c+s #injure
            ' ^vit 0max-! iclose
            i roster - 3 / !wound
         else 2drop
         then
     else 2drop leave ( never any gaps in roster )
     then 3 +loop t;

      dup rot  point>icon

\ do-damage for a particular missle, to ship       (rfg30oct85)
head: ?refresh \ update display only if shields depleted
t:  shield-cnt @ armor-cnt @ null d= drop 1 ( always, for now)
  if refresh-aux then t;

: do-adamage \ damage --- | distributed among resources
   begin ?dup 0= not
     while  #damage (do-damage)
     repeat #wounded @
          if .wounded ' (obits) module ' (role) module
          then ?refresh
  ' %hull-pts ifld@ 0= if end-game then ;

head: do-aldamage
t:  alien-lcalc ship>c do-adamage iclose t;
