HEAD: SCAN-DAMAGE-STATS ( n1 -- [n2] f )
T: >R #SHIP >C+S 0
 7 0 DO %ARMOR I 2* + 1+ C@ J AND
   IF 1+ I SWAP LEAVE THEN
 LOOP R> DROP ICLOSE T;

HEAD: .SHIP-VITALS ( n -- )
T: >R I 0= IF ." DESTROYED" ELSE
      I  1 20 WITHIN IF ." CRITICALLY" ELSE
        I 20 40 WITHIN IF ." HEAVILY" ELSE
          I 40 70 WITHIN IF ." MODERATELY" ELSE
            I 70 90 WITHIN IF ." SLIGHTLY" ELSE
              I 90 100 WITHIN IF ." MINIMALLY" ELSE ." NOT"
 THEN THEN THEN THEN THEN ."  DAMAGED" THEN R> DROP T;

: (/REPAIR) ( -- )
 (/DAMAGE) 0 NCRS ! 0 .REPAIR-CRS 0 .DAMAGE CTINIT
 FTRIG OFF FQUIT OFF #SHIP >C+S
 BEGIN
   XYSCAN DROP NCRS @ OVER - 7 MOD DUP 0< 7 * + NCRS !
   IF OCRS @ .REPAIR-CRS NCRS @ DUP .REPAIR-CRS .DAMAGE THEN
   ?TRIG OCRS @ 2* %ARMOR + >R I 1+ C@ 0> R> %ARMOR = OR AND
   IF OCRS @ .?DESTROYED NOT
     IF 64 SCAN-DAMAGE-STATS
       IF >R I OCRS @ =
         IF I .REPAIRING ELSE I .?NEW-REPAIR
           IF I REPAIR> OCRS @ CTCR >REPAIR
       ELSE I .CONT-REPAIR THEN THEN R> DROP
   ELSE CTERASE OCRS @ >REPAIR THEN THEN THEN ?QUIT
 UNTIL OCRS @ .REPAIR-CRS ICLOSE ;
