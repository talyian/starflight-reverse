CREATE HYPER-DOTS 50 ALLOT   V= ?STAR-TRAIL

: NOVA ( n -- )
 >R 5 20 RRND I C! 0 2 RRND DUP 0=
 72 * 0 2 RRND 2* 1- * -36 36 RRND + -36 MAX 35 MIN I 1+ !
 120 * 0 2 RRND 2* 1- * -60 60 RRND + -60 MAX 59 MIN R> 3 + ! ;

: INIT-HYPER-STARS ( -- )
 50 0 DO HYPER-DOTS I + NOVA 5 +LOOP ;

: HYPER-PLOT ( n -- )
 HYPER-DOTS + >R
 I 3 + @ I C@ / 36 + 0 MAX 71 MIN
 I 1+ @ I C@ / 60 +  0 MAX 119 MIN LPLOT I C@ 1- ?DUP 0=
 IF R> NOVA ELSE R> C! THEN ;

: HYPER-FRAME ( -- )
 ?STAR-TRAIL @ IF DARK THEN
 50 0 DO I HYPER-PLOT 5 +LOOP V>DISPLAY ;

: FLUX-EFFECT ( -- )
 >MAINVIEW 1 ?STAR-TRAIL ! INIT-HYPER-STARS
 100 0 DO HYPER-FRAME
 LOOP DARK V>DISPLAY >DIS~~AY ;
