---- Screen 9 ----
( Bank  TRANSACT                                  AWK 10/23/85 )

: TRANSACT ( n -- )
 SET-BANK TFLAG C@
 IF IOPEN 14 SWAP 1 *CREATE IFIRST
   0 9 0 DO INEXT ?FIRST
     IF 1+ LEAVE THEN
   LOOP 0=
   IF IFIRST IDELETE THEN ILAST COVER SET-CURRENT
     STARDATE @ MASTERBALANCE D@ BALANCE D@ D-
   ICLOSE T-AMT D! T-DATE ! ICLOSE
   BALANCE D@ MASTERBALANCE D! 0 TFLAG C!
 ELSE DROP THEN ICLOSE ;
STARDATE OFF
SET-BANK 100080. MASTERBALANCE D! 100000. BALANCE D! 1 TFLAG C!
ICLOSE 6 TRANSACT

: INT% ( -- )
 INIT-BALANCE
 D@BALANCE 12 100 M*/             \ interest/annum
 STARDATE @ PORTDATE @ - 300 M*/  \ adjusted for time
 2DUP D0=
 IF 2DROP
 ELSE D@BALANCE D+ D!BALANCE      \ modify balance
   !TFLAG 5 TRANSACT              \ note transaction
   STARDATE @ PORTDATE !          \ set latest interest date
 THEN ;
